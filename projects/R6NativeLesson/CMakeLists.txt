cmake_minimum_required(VERSION 3.25)
project(R6NativeLesson LANGUAGES CXX)
if(NOT TARGET PipeFrame::Engine)
  if(DEFINED PIPEFRAME_ENGINE_LIBRARY AND DEFINED PIPEFRAME_SDK_ROOT)
    add_library(PipeFrame::Engine SHARED IMPORTED)
    set_target_properties(PipeFrame::Engine PROPERTIES IMPORTED_LOCATION "${PIPEFRAME_ENGINE_LIBRARY}" INTERFACE_INCLUDE_DIRECTORIES "${PIPEFRAME_SDK_ROOT}/engine/include")
  elseif(DEFINED PIPEFRAME_SDK_ROOT)
    add_subdirectory("${PIPEFRAME_SDK_ROOT}/engine" "${CMAKE_BINARY_DIR}/PipeFrameEngine")
  else()
    find_package(PipeFrame CONFIG REQUIRED)
  endif()
endif()
file(GLOB_RECURSE PROJECT_SOURCES CONFIGURE_DEPENDS Source/*.cpp Source/*.h Source/*.hpp)
add_library(R6NativeLessonRuntime SHARED ${PROJECT_SOURCES})
target_link_libraries(R6NativeLessonRuntime PRIVATE PipeFrame::Engine)
target_include_directories(R6NativeLessonRuntime PRIVATE Source)
target_compile_definitions(R6NativeLessonRuntime PRIVATE PIPEFRAME_BUILDING_PROJECT_RUNTIME)
target_compile_features(R6NativeLessonRuntime PRIVATE cxx_std_20)
set_target_properties(R6NativeLessonRuntime PROPERTIES PREFIX "" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/Build/$<CONFIG>" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/Build/$<CONFIG>")
